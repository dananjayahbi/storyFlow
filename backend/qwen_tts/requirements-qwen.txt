#   uv venv --python 3.11 backend\qwen_tts\.venv
#   uv pip install --python backend\qwen_tts\.venv\Scripts\python.exe `
#       "torch==2.6.0+cu124" "torchaudio==2.6.0+cu124" `
#       --index-url https://download.pytorch.org/whl/cu124 --index-strategy unsafe-best-match
#   uv pip install --python backend\qwen_tts\.venv\Scripts\python.exe -r backend\qwen_tts\requirements-qwen.txt
#
